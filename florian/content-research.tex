% ========================== RESEARCH INTERESTS ============================
\section{Research Interests}

{\bodysize
Interpretability of language models, and what can legitimately be inferred from probing their internal representations. Current work concerns cross-lingual alignment in small Japanese--English bilingual models: whether pragmatic and cultural distinctions such as \textit{keigo} are genuinely encoded, or reconstructed from surface cues. Related interest in evaluation methodology --- benchmark contamination, and the gap between what intrinsic multilingual evaluations claim to measure and what they measure.
}

\section{Research \& Technical Projects}

\project{Predicting Player Progression from On-Pitch Performance Data \,{\normalfont\color{subtle}--- OH Leuven International Week}}{Mar 2026}
\begin{bullets}
    \item Built a predictive model for whether a lower-league goalkeeper will succeed at a higher level, using performance data alone and no contextual variables.
    \item Dataset of 693 goalkeepers across 40+ leagues (2019--2026); engineered 1{,}009 KPIs per player and reduced to 75 through a six-method feature selection pipeline.
    \item Random Forest + XGBoost ensemble under 5-fold cross-validation reached 0.87 AUC; passing and distribution, not shot-stopping, were the strongest predictors of progression.
\end{bullets}

\project{Decoding EEG Signals to Interpretable Outputs \,{\normalfont\color{subtle}--- NeuroMedix Hackathon, finalist}}{Mar 2026}
\begin{bullets}
    \item Applied machine learning to raw EEG recordings, translating decoded neural signals into interpretable outputs on a sponsor-defined case in a team of four.
\end{bullets}

\project{On-Device Behavioural Modelling for Burnout Detection \,{\normalfont\color{subtle}--- CalmCampus}}{Jan 2026 -- Present}
\begin{bullets}
